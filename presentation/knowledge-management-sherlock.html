Step 3A · Serve (BigQuery Hybrid Search + BigQuery Graph)

Combine vector and keyword search so rare names never get missed

Initial passage retrieval seeds the graph query — fusing dense vectors with lexical search guarantees exact entity recall.

-- 1. Hybrid Search: Vector + Full-Text Reciprocal Rank Fusion
CREATE TEMP TABLE hits AS
SELECT passage_id, story_id, chapter_num, content,
  (1.0 / (60 + sem_rank)) + (1.0 / (60 + lex_rank)) AS rrf_score
FROM semantic_hits FULL JOIN lexical_hits USING (passage_id)
ORDER BY rrf_score DESC LIMIT 8;
-- 2. Multi-Hop Context Expansion via BigQuery Graph (ISO GQL)
SELECT g.* FROM GRAPH_TABLE(
  `sherlock.story_graph`
  MATCH (c:Character)-[p:PARTICIPATED_IN]->(e:Event)
  OPTIONAL MATCH (e)-[:OCCURRED_AT]->(l:Location)
  RETURN e.event_name, c.canonical_name, p.role, l.canonical_name
) g JOIN hit_events USING (event_id);
